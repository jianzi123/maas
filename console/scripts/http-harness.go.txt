package controlapi
import("context";"crypto/sha256";"database/sql";"encoding/hex";"encoding/json";"github.com/go-sql-driver/mysql";"github.com/google/uuid";"github.com/jianzi123/maas/internal/kernel";"net/http/httptest";"os";"os/exec";"strings";"testing";"time")
func TestFrontendHTTP(t *testing.T) {
	socket := os.Getenv("MAAS_MYSQL_TEST_SOCKET")
	if socket == "" {
		t.Skip("disposable MySQL required")
	}
	if os.Getenv("MAAS_DISPOSABLE_MYSQL") != "1" || !strings.HasPrefix(socket, "/tmp/maas11-mysql.") {
		t.Fatal("refuse non-disposable DB")
	}
	cfg := mysql.NewConfig()
	cfg.User = "root"
	cfg.Net = "unix"
	cfg.Addr = socket
	cfg.ParseTime = true
	cfg.Loc = time.UTC
	cfg.Timeout = 2 * time.Second
	cfg.ReadTimeout = 5 * time.Second
	cfg.WriteTimeout = 5 * time.Second
	admin, e := sql.Open("mysql", cfg.FormatDSN())
	if e != nil {
		t.Fatal(e)
	}
	defer admin.Close()
	name := "maas_api_test_" + strings.ReplaceAll(uuid.NewString(), "-", "")
	if _, e = admin.Exec("CREATE DATABASE `" + name + "`"); e != nil {
		t.Fatal(e)
	}
	defer admin.Exec("DROP DATABASE `" + name + "`")
	cfg.DBName = name
	db, _ := sql.Open("mysql", cfg.FormatDSN())
	defer db.Close()
	ctx := context.Background()
	if e = kernel.Migrate(ctx, db); e != nil {
		t.Fatal(e)
	}
	core, _ := kernel.New(db, "controlled-api", []byte(strings.Repeat("p", 32)), nil, nil)
	if e = core.Configure(ctx, kernel.RatePolicy{Revision: "test-rate-v1", Epoch: 1, WindowMS: 3600000, GlobalLimit: 100, KeyLimit: 100}, 1); e != nil {
		t.Fatal(e)
	}
	session, csrf := uuid.NewString()+uuid.NewString(), uuid.NewString()+uuid.NewString()
	digest := func(s string) string { x := sha256.Sum256([]byte(s)); return hex.EncodeToString(x[:]) }
	h := Handler{Core: core, Sessions: BootstrapSession{digest(session), digest(csrf), "https://console.test", time.Now().Add(time.Minute)}}
	server := httptest.NewServer(h)
	defer server.Close()
 nodeBinary, consoleDir, nodeConsoleDir := os.Getenv("MAAS_FE_NODE"), os.Getenv("MAAS_FE_CONSOLE_DIR"), os.Getenv("MAAS_FE_CONSOLE_NODE_DIR")
 if nodeBinary == "" || consoleDir == "" || nodeConsoleDir == "" { t.Fatal("explicit temporary frontend test paths required") }
 cmd := exec.Command(nodeBinary, nodeConsoleDir+"/node_modules/vitest/vitest.mjs", "run", "src/http-continuation.test.ts", "--reporter=json", "--outputFile=artifacts/http-continuation.json")
 cmd.Dir = consoleDir
 cmd.Env = append(os.Environ(), "MAAS_HTTP_URL="+server.URL, "MAAS_HTTP_SESSION="+session, "MAAS_HTTP_CSRF="+csrf, "MAAS_HTTP_ORIGIN=https://console.test", "WSLENV=MAAS_HTTP_URL:MAAS_HTTP_SESSION:MAAS_HTTP_CSRF:MAAS_HTTP_ORIGIN")
 if out, err := cmd.CombinedOutput(); err != nil {t.Fatalf("frontend HTTP consumers failed: %v\n%s", err, out)} else {t.Log(string(out))}
 report, err := os.ReadFile(cmd.Dir+"/artifacts/http-continuation.json")
 if err != nil { t.Fatal(err) }
 var result struct { NumPassedTests int; NumFailedTests int; NumPendingTests int }
 if json.Unmarshal(report, &result) != nil || result.NumPassedTests != 11 || result.NumFailedTests != 0 || result.NumPendingTests != 0 { t.Fatalf("HTTP checks did not execute fully: %s", report) }
}
