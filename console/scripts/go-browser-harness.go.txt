package controlapi

import (
	"context"
	"crypto/sha256"
	"database/sql"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"github.com/go-sql-driver/mysql"
	"github.com/google/uuid"
	"github.com/jianzi123/maas/internal/kernel"
	"net"
	"net/http/httptest"
	"os"
	"os/exec"
	"strings"
	"testing"
	"time"
)

func TestFrontendBrowser(t *testing.T) {
	socket := os.Getenv("MAAS_MYSQL_TEST_SOCKET")
	if socket == "" {
		t.Skip("disposable MySQL required")
	}
	if os.Getenv("MAAS_DISPOSABLE_MYSQL") != "1" || !strings.HasPrefix(socket, "/tmp/maas11-mysql.") {
		t.Fatal("refuse non-disposable DB")
	}
	cfg := mysql.NewConfig()
	cfg.User = "root"
	cfg.Net = "unix"
	cfg.Addr = socket
	cfg.ParseTime = true
	cfg.Loc = time.UTC
	cfg.Timeout = 2 * time.Second
	cfg.ReadTimeout = 5 * time.Second
	cfg.WriteTimeout = 5 * time.Second
	admin, err := sql.Open("mysql", cfg.FormatDSN())
	if err != nil {
		t.Fatal(err)
	}
	defer admin.Close()
	name := "maas_api_test_" + strings.ReplaceAll(uuid.NewString(), "-", "")
	if _, err = admin.Exec("CREATE DATABASE " + name); err != nil {
		t.Fatal(err)
	}
	defer admin.Exec("DROP DATABASE " + name)
	cfg.DBName = name
	db, err := sql.Open("mysql", cfg.FormatDSN())
	if err != nil {
		t.Fatal(err)
	}
	defer db.Close()
	if err = kernel.Migrate(context.Background(), db); err != nil {
		t.Fatal(err)
	}
	core, err := kernel.New(db, "controlled-api", []byte(strings.Repeat("p", 32)), nil, nil)
	if err != nil {
		t.Fatal(err)
	}
	if err = core.Configure(context.Background(), kernel.RatePolicy{Revision: "test-rate-v1", Epoch: 1, WindowMS: 3600000, GlobalLimit: 100, KeyLimit: 100}, 1); err != nil {
		t.Fatal(err)
	}
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	port := listener.Addr().(*net.TCPAddr).Port
	listener.Close()
	session, csrf := uuid.NewString()+uuid.NewString(), uuid.NewString()+uuid.NewString()
	digest := func(s string) string { x := sha256.Sum256([]byte(s)); return hex.EncodeToString(x[:]) }
	origin := fmt.Sprintf("http://127.0.0.1:%d", port)
	expires := time.Now().Add(5 * time.Minute)
	server := httptest.NewServer(Handler{Core: core, Sessions: BootstrapSession{digest(session), digest(csrf), origin, expires}})
	defer server.Close()
	node, dir, nodeDir := os.Getenv("MAAS_FE_NODE"), os.Getenv("MAAS_FE_CONSOLE_DIR"), os.Getenv("MAAS_FE_CONSOLE_NODE_DIR")
	if node == "" || dir == "" || nodeDir == "" {
		t.Fatal("explicit frontend paths required")
	}
	command := exec.Command(node, nodeDir+"/scripts/go-browser.mjs")
	command.Dir = dir
	command.Env = append(os.Environ(), "MAAS_LOCAL_API="+server.URL, "MAAS_LOCAL_SESSION="+session, "MAAS_LOCAL_CSRF="+csrf, "MAAS_LOCAL_EXPIRES="+expires.Format(time.RFC3339), "MAAS_LOCAL_PORT="+fmt.Sprint(port), "WSLENV=MAAS_LOCAL_API:MAAS_LOCAL_SESSION:MAAS_LOCAL_CSRF:MAAS_LOCAL_EXPIRES:MAAS_LOCAL_PORT")
	output, err := command.CombinedOutput()
	if err != nil {
		t.Fatalf("Go browser checks failed: %v\n%s", err, output)
	}
	t.Log(string(output))
	raw, err := os.ReadFile(dir + "/artifacts/go-browser/results.json")
	if err != nil {
		t.Fatal(err)
	}
	var report struct {
		Checks []string
		Errors []string
	}
	if json.Unmarshal(raw, &report) != nil || len(report.Checks) != 14 || len(report.Errors) != 0 {
		t.Fatal("all 14 Go browser checks must execute")
	}
}
