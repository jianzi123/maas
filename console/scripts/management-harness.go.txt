func TestWorkspaceFrontendBrowser(t *testing.T) {
	socket := os.Getenv("MAAS_MYSQL_TEST_SOCKET")
	if os.Getenv("MAAS_DISPOSABLE_MYSQL") != "1" || !strings.HasPrefix(socket, "/tmp/maas11-mysql.") || os.Getenv("KUBEBUILDER_ASSETS") == "" {
		t.Fatal("explicit disposable MySQL and envtest assets required")
	}
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	cfg := mysql.NewConfig()
	cfg.User = "root"
	cfg.Net = "unix"
	cfg.Addr = socket
	cfg.ParseTime = true
	cfg.Loc = time.UTC
	cfg.Timeout = 3 * time.Second
	admin, e := sql.Open("mysql", cfg.FormatDSN())
	if e != nil {
		t.Fatal(e)
	}
	defer admin.Close()
	database := "maas_operations_" + strings.ReplaceAll(uuid.NewString(), "-", "")
	if _, e = admin.Exec("CREATE DATABASE `" + database + "`"); e != nil {
		t.Fatal(e)
	}
	defer admin.Exec("DROP DATABASE `" + database + "`")
	cfg.DBName = database
	db, e := sql.Open("mysql", cfg.FormatDSN())
	if e != nil {
		t.Fatal(e)
	}
	defer db.Close()
	if e = kernel.Migrate(ctx, db); e != nil {
		t.Fatal(e)
	}
	if e = mysqlstore.Migrate(ctx, db); e != nil {
		t.Fatal(e)
	}
	core, e := kernel.New(db, "management-envtest", []byte(strings.Repeat("p", 32)), nil, nil)
	if e != nil {
		t.Fatal(e)
	}
	if e = core.Configure(ctx, kernel.RatePolicy{Revision: "test-only", Epoch: 1, WindowMS: 1000, GlobalLimit: 100, KeyLimit: 100}, 2); e != nil {
		t.Fatal(e)
	}
	publisher, e := mysqlstore.New(db, "management-envtest", core)
	if e != nil {
		t.Fatal(e)
	}
	publisher = publisher.WithManagementOperations()
	existing := false
	environment := &envtest.Environment{UseExistingCluster: &existing, CRDDirectoryPaths: []string{filepath.Join("..", "..", "config", "crd")}, ErrorIfCRDPathMissing: true}
	kubeConfig, e := environment.Start()
	if e != nil {
		t.Fatal(e)
	}
	defer func() {
		if e := environment.Stop(); e != nil {
			t.Error(e)
		}
	}()
	scheme := runtime.NewScheme()
	_ = api.AddToScheme(scheme)
	_ = corev1.AddToScheme(scheme)
	kube, e := client.New(kubeConfig, client.Options{Scheme: scheme})
	if e != nil {
		t.Fatal(e)
	}
	ns := "maas-models-example"
	if e = kube.Create(ctx, &corev1.Namespace{ObjectMeta: metav1.ObjectMeta{Name: ns}}); e != nil {
		t.Fatal(e)
	}
	access := history.NamespaceAccess{Namespaces: map[string]bool{ns: true}, Domains: map[string]bool{"test-domain": true}, ScopeID: "test-session"}
	b, e := os.ReadFile("../../config/samples/MAAS-2-modelrelease-externalapi-v0.4.example.yaml")
	if e != nil {
		t.Fatal(e)
	}
	original := new(api.ModelRelease)
	if e = yaml.Unmarshal(b, original); e != nil {
		t.Fatal(e)
	}
	original.Name = "managed-model"
	resolution := release.Resolution{Versions: map[string]string{"authorization_domain": "test-domain"}, GatewayNamespace: ns, IngressClass: "higress", HostSuffix: "test.invalid", EndpointDNS: "fixture.invalid", EndpointPort: 443, CredentialHandle: "opaque-test-only", PluginDigest: strings.Repeat("1", 64)}
	updated := original.DeepCopy()
	updated.Spec.Source.ExternalAPI.UpstreamModel = "test-second-model"
	registry := &management.Registry{Access: access, Entries: []management.RegistryEntry{{Namespace: ns, Spec: original.Spec, Resolution: resolution}, {Namespace: ns, Spec: updated.Spec, Resolution: resolution}}}
	hr := &history.Reader{DB: db, PayloadDB: db, Cluster: "management-envtest", Access: access, CursorKey: []byte(strings.Repeat("c", 32)), Live: history.KubeLive{Reader: kube}}
	hr.Evaluator = history.ResolverEvaluator{Resolver: registry, Loader: hr}
	op := &management.Operations{DB: db, Cluster: "management-envtest", Access: access}
	service := &management.Service{Client: kube, Reader: kube, Operations: op, History: hr, Resolver: registry, Access: access, CursorKey: hr.CursorKey}

	listener, e := net.Listen("tcp", "127.0.0.1:0")
	if e != nil {
		t.Fatal(e)
	}
	port := listener.Addr().(*net.TCPAddr).Port
	listener.Close()
	session, csrf := uuid.NewString()+uuid.NewString(), uuid.NewString()+uuid.NewString()
	digest := func(s string) string { x := sha256.Sum256([]byte(s)); return hex.EncodeToString(x[:]) }
	expires := time.Now().Add(5 * time.Minute)
	handler := controlapi.Handler{Core: core, Sessions: controlapi.BootstrapSession{SessionSHA256: digest(session), CSRFSHA256: digest(csrf), Origin: fmt.Sprintf("http://127.0.0.1:%d", port), Expires: expires}, Releases: service, Revisions: hr, Models: core, ModelGate: publisher, ModelNamespaces: access.Namespaces}
	server := httptest.NewServer(handler)
	defer server.Close()
	reconciler := &release.Reconciler{Client: kube, Reader: kube, Dependencies: release.Dependencies{Control: publisher, Inventory: publisher, Resolver: registry, Bundle: noGatewayBundle{}, Plugin: noGatewayProducer{}}}
	stopped := make(chan struct{})
	workerErrors := make(chan error, 1)
	registered := map[string]bool{}
	passes := map[string]int{}
	go func() {
		defer close(stopped)
		ticker := time.NewTicker(150 * time.Millisecond)
		defer ticker.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
				var list api.ModelReleaseList
				if err := kube.List(ctx, &list, client.InNamespace(ns)); err != nil {
					if ctx.Err() == nil {
						select {
						case workerErrors <- err:
						default:
						}
					}
					return
				}
				for _, m := range list.Items {
					key := fmt.Sprint(m.UID, "/", m.Generation, "/", m.DeletionTimestamp)
					if passes[key] < 4 {
						passes[key]++
						if _, err := reconciler.Reconcile(ctx, ctrl.Request{NamespacedName: client.ObjectKeyFromObject(&m)}); err != nil && ctx.Err() == nil {
							select {
							case workerErrors <- err:
							default:
							}
						}
					}
					{
						state, err := publisher.Read(ctx, string(m.UID))
						if err == nil && state.Candidate != nil && !registered[state.Candidate.PublicationID] {
							profile := kernel.Profile{CRUID: string(m.UID), Namespace: ns, Name: m.Name, Alias: m.Spec.ModelAlias, Binding: *state.Candidate, SourceID: uuid.NewString(), SourceType: "external", UpstreamModel: m.Spec.Source.ExternalAPI.UpstreamModel, ConfigRevision: "test-config", CapabilityRevision: "test-capability", RouteDigest: strings.Repeat("2", 64), GatewayEpoch: 1, ApprovalRef: "test-only-metadata", Catalog: &kernel.CatalogMetadata{DisplayName: "本地发布验证模型", Capabilities: map[string]any{"streaming": true, "verified_max_output_tokens": 100, "temperature_supported": false, "usage_available": "unknown", "capability_revision": "test-capability", "hard_budget_status": "unverified", "budget_unavailable_reason": "尚无真实网关和验证证明", "protocols": []any{"chat_completions"}}}}
							if err := core.RegisterProfile(ctx, profile); err == nil {
								registered[state.Candidate.PublicationID] = true
							}
						}
					}
				}
			}
		}
	}()
	defer func() { cancel(); <-stopped }()
	node, dir, nodeDir := os.Getenv("MAAS_FE_NODE"), os.Getenv("MAAS_FE_CONSOLE_DIR"), os.Getenv("MAAS_FE_CONSOLE_NODE_DIR")
	if node == "" || dir == "" || nodeDir == "" {
		t.Fatal("explicit frontend paths required")
	}
	input, _ := json.Marshal(map[string]any{"apiVersion": api.GroupVersion.String(), "kind": "ModelRelease", "metadata": map[string]any{"namespace": original.Namespace, "name": original.Name}, "spec": original.Spec})
	command := exec.Command(node, nodeDir+"/scripts/management-browser.mjs")
	command.Dir = dir
	command.Env = append(os.Environ(), "MAAS_LOCAL_API="+server.URL, "MAAS_LOCAL_SESSION="+session, "MAAS_LOCAL_CSRF="+csrf, "MAAS_LOCAL_EXPIRES="+expires.Format(time.RFC3339), "MAAS_LOCAL_PORT="+fmt.Sprint(port), "MAAS_BROWSER_CR="+string(input), "WSLENV=MAAS_LOCAL_API:MAAS_LOCAL_SESSION:MAAS_LOCAL_CSRF:MAAS_LOCAL_EXPIRES:MAAS_LOCAL_PORT:MAAS_BROWSER_CR")
	out, err := command.CombinedOutput()
	if err != nil {
		t.Fatalf("frontend failed: %v\n%s", err, out)
	}
	t.Log(string(out))
	select {
	case err := <-workerErrors:
		t.Fatal(err)
	default:
	}
	raw, err := os.ReadFile(dir + "/artifacts/management-browser/results.json")
	if err != nil {
		t.Fatal(err)
	}
	var report struct {
		Checks []string
		Errors []string
	}
	if json.Unmarshal(raw, &report) != nil || len(report.Checks) < 12 || len(report.Errors) != 0 {
		t.Fatal("management browser checks incomplete")
	}
}
